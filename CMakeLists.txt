cmake_minimum_required(VERSION 3.24)
project(STLCutStudioAI VERSION 0.3.2 LANGUAGES C CXX)
if(MSVC)
    if(CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "^Nota: inclusi")
        # The local Spanish-only compiler emits UTF-8; CMake's probe can decode
        # its non-ASCII prefix as OEM. Normalize the observed prefix for Ninja.
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "Nota: inclusión del archivo: ")
    elseif(NOT CMAKE_CL_SHOWINCLUDES_PREFIX STREQUAL "Note: including file: ")
        message(FATAL_ERROR "Prefijo /showIncludes no reconocido; verificar dependencias de cabeceras antes de compilar.")
    endif()
endif()

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Esta base requiere Windows y MSVC x64.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure un entorno MSVC x64.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

# Kit oficial instalado; build Release local verificado. Véase docs/WINDOWS_SETUP.md.
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets OpenGL OpenGLWidgets)
get_target_property(qt_core_type Qt6::Core TYPE)
if(NOT qt_core_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "Esta configuracion requiere Qt dinamico.")
endif()

qt_standard_project_setup()
set(MANIFOLD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/manifold-3.5.4/manifold-3.5.4")
if(NOT EXISTS "${MANIFOLD_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Ejecute scripts/bootstrap-manifold.ps1 para instalar Manifold 3.5.4 verificado.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Manifold static Windows library" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "No test infrastructure" FORCE)
foreach(option MANIFOLD_TEST MANIFOLD_PAR MANIFOLD_CROSS_SECTION MANIFOLD_CBIND
               MANIFOLD_PYBIND MANIFOLD_JSBIND MANIFOLD_DOWNLOADS)
    set(${option} OFF CACHE BOOL "Minimal pinned Manifold build" FORCE)
endforeach()
add_subdirectory("${MANIFOLD_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/manifold" EXCLUDE_FROM_ALL)
set_target_properties(manifold PROPERTIES AUTOMOC OFF AUTOUIC OFF)
set(LIB3MF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/lib3mf-2.5.0/lib3mf_sdk")
set(MINIZ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/miniz-3.1.0")
if(NOT EXISTS "${MINIZ_ROOT}/miniz.c")
    message(FATAL_ERROR "Ejecute scripts/bootstrap-miniz.ps1 para instalar miniz 3.1.0 verificado.")
endif()
add_library(stlcut_miniz STATIC "${MINIZ_ROOT}/miniz.c" "${MINIZ_ROOT}/miniz_tinfl.c" "${MINIZ_ROOT}/miniz_zip.c")
set_target_properties(stlcut_miniz PROPERTIES AUTOMOC OFF AUTOUIC OFF)
include(GenerateExportHeader)
generate_export_header(stlcut_miniz BASE_NAME MINIZ EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h")
target_include_directories(stlcut_miniz PUBLIC "${MINIZ_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(stlcut_miniz PUBLIC MINIZ_STATIC_DEFINE MINIZ_NO_STDIO MINIZ_NO_TIME
    MINIZ_NO_DEFLATE_APIS MINIZ_NO_ZLIB_APIS MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
if(NOT EXISTS "${LIB3MF_ROOT}/Lib/lib3mf.lib")
    message(FATAL_ERROR "Ejecute scripts/bootstrap-lib3mf.ps1 para instalar lib3mf 2.5.0 verificado.")
endif()
add_library(lib3mf SHARED IMPORTED)
set_target_properties(lib3mf PROPERTIES
    IMPORTED_IMPLIB "${LIB3MF_ROOT}/Lib/lib3mf.lib"
    IMPORTED_LOCATION "${LIB3MF_ROOT}/Bin/lib3mf.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${LIB3MF_ROOT}/Bindings/Cpp")
add_library(stlcut_core STATIC core/io/StlReader.cpp core/io/MeshReader.cpp
    core/io/StlWriter.cpp core/io/ZipParts.cpp core/io/PaintCodec.cpp core/cutting/PlaneCut.cpp core/cutting/ColorSplit.cpp core/projects/EditHistory.cpp)
target_link_libraries(stlcut_core PUBLIC lib3mf PRIVATE manifold stlcut_miniz)
target_compile_definitions(stlcut_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(stlcut_core PROPERTIES AUTOMOC OFF AUTOUIC OFF)
target_include_directories(stlcut_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(stlcut_core PRIVATE /W4 /permissive- /utf-8)
qt_add_executable(STLCutStudioAI WIN32 apps/desktop/main.cpp apps/desktop/MeshView.cpp apps/desktop/MeshView.h apps/desktop/ModelImport.cpp apps/desktop/ModelImport.h)
target_link_libraries(STLCutStudioAI PRIVATE Qt6::Widgets Qt6::OpenGLWidgets Qt6::OpenGL stlcut_core)
target_compile_options(STLCutStudioAI PRIVATE /W4 /permissive- /utf-8)
target_compile_definitions(STLCutStudioAI PRIVATE
    STL_CUT_STUDIO_VERSION="${PROJECT_VERSION}")
install(TARGETS STLCutStudioAI RUNTIME DESTINATION .)
add_custom_command(TARGET STLCutStudioAI POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${LIB3MF_ROOT}/Bin/lib3mf.dll" "$<TARGET_FILE_DIR:STLCutStudioAI>/lib3mf.dll")
install(FILES "${LIB3MF_ROOT}/Bin/lib3mf.dll" DESTINATION .)
install(FILES "${LIB3MF_ROOT}/License.txt" DESTINATION licenses/lib3mf-2.5.0)
install(FILES "${MINIZ_ROOT}/LICENSE" DESTINATION licenses/miniz-3.1.0)
install(FILES "${MANIFOLD_ROOT}/LICENSE" "${MANIFOLD_ROOT}/AUTHORS" DESTINATION licenses/manifold-3.5.4)
